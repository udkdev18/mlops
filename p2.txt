import threading
import time
import joblib
import requests
from flask import Flask, request, jsonify
from sklearn.datasets import load_diabetes
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error

# Train and save the model
data = load_diabetes()
X_train, X_test, y_train, y_test = train_test_split(
    data.data, data.target, test_size=0.2, random_state=42
)

model = LinearRegression().fit(X_train, y_train)
joblib.dump(model, "diabetes_model.joblib")
print(f"Mean Squared Error: {mean_squared_error(y_test, model.predict(X_test)):.2f}")

# Create the Flask prediction API
app = Flask(__name__)
loaded_model = joblib.load("diabetes_model.joblib")

@app.get("/")
def home():
    return jsonify(status="API is running")

@app.post("/predict")
def predict():
    features = request.json["features"]
    prediction = loaded_model.predict([features])[0]
    return jsonify(prediction=float(prediction))

# Start the API and request a prediction
threading.Thread(
    target=lambda: app.run(port=5000, use_reloader=False),
    daemon=True
).start()

time.sleep(2)
response = requests.post(
    "http://127.0.0.1:5000/predict",
    json={"features": X_test[0].tolist()}
)
print(response.json())
